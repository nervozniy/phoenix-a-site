<?php


function greet(...$args) 
{
    if (count($args) === 1) {
        $receiver = $args[0];
        $for = [
            'Король' => 'Добрый день, Ваше Величество',
            'друг' => 'Привет, Серёжа!'
        ];
        return $for[$receiver] ?? "Привет, $receiver";
    }

    $key = $args[0] . '->' . $args[1];
    $foreach = [
        'Штурмовик->Дарт Вейдер' => 'Император здрасте',
        'Дарт Вейдер->Штурмовик' => 'Кивок, штурмовик'
    ];
    return $foreach[$key] ?? "Приветствие не найдено";
}
// Проверка
echo greet('Король') . "\n";
echo greet('друг') . "\n";
echo greet('Штурмовик', 'Дарт Вейдер') . "\n";
echo greet('Дарт Вейдер', 'Штурмовик') . "\n";
